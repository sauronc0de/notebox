cmake_minimum_required(VERSION 3.16)
project(notebox VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.2.0 REQUIRED)
add_executable(notebox src/main.cpp)
target_include_directories(notebox PRIVATE third_party)
target_link_libraries(notebox PRIVATE SQLite::SQLite3 nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(notebox PRIVATE -Wall -Wextra -Wpedantic)
